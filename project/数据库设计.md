#### 好友关系表（`Friendship`）

```python
from django.db import models
from django.contrib.auth import get_user_model
from django.db.models import Q, F

class Friendship(models.Model):
    """记录两个用户之间的好友关系（无方向，A-B 与 B-A 视为同一条记录）"""
    user_a = models.ForeignKey(
        get_user_model(),
        on_delete=models.CASCADE,
        related_name='friendships_a',
        db_index=True
    )
    user_b = models.ForeignKey(
        get_user_model(),
        on_delete=models.CASCADE,
        related_name='friendships_b',
        db_index=True
    )
    created_at = models.DateTimeField(auto_now_add=True)

    class Meta:
        constraints = [
            # 防止 self-friend
            models.CheckConstraint(condition=~Q(user_a=F('user_b')), name='friendship_no_self'),
            # 确保 (user_a, user_b) 唯一（需在 save 中把顺序规范化）
            models.UniqueConstraint(fields=['user_a', 'user_b'], name='unique_friendship'),
        ]

    def save(self, *args, **kwargs):
        if self.user_a_id and self.user_b_id:
            if self.user_a_id == self.user_b_id:
                raise ValueError("A user cannot befriend themselves.")
            if self.user_a_id > self.user_b_id:
                self.user_a, self.user_b = self.user_b, self.user_a
        super().save(*args, **kwargs)

    def __str__(self):
        return f"{self.user_a} ↔ {self.user_b}"
    
class Pending(models.Model):
    user_from = models.ForeignKey(
        get_user_model(),
        on_delete=models.CASCADE,
        related_name="user_from",
        db_index=True
    )
    user_to = models.ForeignKey(
        get_user_model(),
        on_delete=models.CASCADE,
        related_name="user_to",
        db_index=True
    )
    created_at = models.DateTimeField(auto_now_add=True)
    
    def __str__(self):
        return f"{self.user_from} → {self.user_to}"
```

```python
    def __str__(self):
```

### 2. **在线会话通用部分**

#### 会话表（`Conversation`）

记录所有会话（包括私聊、群聊等）的基本信息：
**有个多态的想法，对于一个会话的信息展示，有一部分信息是相同的，有一部分是不同的。对于不同的部分可以封装对外函数，在内部通过检查type采用不同实现。**
**不同**：
1. 名称（check_name）(头像可以类似，甚至直接一起,check_name_and_avatar)
- 对私聊：返回好友名称。实现方式是先取出Conversation的members,其中不为当前用户的成员的username;
- 对群聊: return Conversation.name




```python
class Conversation(models.Model):
    conversation_id = models.AutoField(primary_key=True)  # 会话ID
    type = models.CharField(max_length=10, choices=[ # 会话种类（群聊/私聊）
        ("private", "Private"), ("group", "Group")
    ])
    created_at = models.DateTimeField(auto_now_add=True)  # 创建时间

    name = models.CharField(max_length=100, blank=True, null=True)  # 会话名称（群聊）
    avatar_url = models.CharField(max_length=1024, blank=True, null=True) # 会话头像（群聊）

    def __str__(self):
        return self.name or f'Private chat with {self.creator.username}'
```

```python
class Message(models.Model):
    message_id = models.AutoField(primary_key=True)  # 消息ID
    conversation = models.ForeignKey(Conversation, on_delete=models.CASCADE, , related_name='messages')  # 所属会话
    sender = models.ForeignKey(User, on_delete=models.CASCADE, related_name='sent_messages')  # 发送者
    content = models.TextField()  # 消息内容

    sent_at = models.DateTimeField(auto_now_add=True)  # 发送时间
    is_edited = models.BooleanField(default=False)  # 是否已编辑
    is_recalled = models.BooleanField(default=False)  # 是否已撤回（撤回或者也可以直接删除该条聊天记录？）

    deleted_by = models.ForeignKey(User, on_delete=models.CASCADE, related_name='deleted_messages') # 这样用户可以通过deleted_messages快速筛选已经删除的消息

    reply_count = models.IntegerField(default=0)  # 回复该消息的消息数量
    replied_message = models.ForeignKey('self', null=True, blank=True, on_delete=models.SET_NULL, related_name='replies')  # 被回复的消息（这个地方如何处理已经撤回的回复消息应该要根据撤回消息的实现？）
    # 一种可能的办法，标记回复的消息是否已删除或已撤回？
    is_reply_to_deleted_message = models.BooleanField(default=False)  # 是否回复的消息已删除
    is_reply_to_recalled_message = models.BooleanField(default=False)  # 是否回复的消息已撤回

    def __str__(self):
        return f"{self.sender.username} sent a {self.message_type} message"
```

```python
class GroupAnnouncement(models.Model):
    conversation = models.ForeignKey(Conversation, on_delete=models.CASCADE)  # 群聊
    posted_by = models.ForeignKey(User, on_delete=models.CASCADE)  # 发布者（群主或管理员）
    content = models.TextField()  # 公告内容
    posted_at = models.DateTimeField(auto_now_add=True)  # 发布时间

    def __str__(self):
        return f"Announcement by {self.posted_by.username} on {self.posted_at}"
```
